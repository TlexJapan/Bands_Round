metroBeat=(metroBeat+1)%count; } // スタート async function startMetro(){ if(metroTimer!==null)return; if(!metroContext){ metroContext=new(window.AudioContext||window.webkitAudioContext)(); } await metroContext.resume(); metroBeat=0; tick(); scheduleNextBeat(); } document.getElementById("startMetro").onclick=startMetro; // ストップ document.getElementById("stopMetro").onclick=()=>{ if(metroTimer!==null){ clearTimeout(metroTimer);
